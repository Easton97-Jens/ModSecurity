cmake_minimum_required(VERSION 3.16)
project(modsecurity_blaze_wrapper LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

get_filename_component(MODSECURITY_SOURCE_DIR
  "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

set(BLAZE_COMPILER ON CACHE BOOL "" FORCE)
set(BLAZE_EVALUATOR ON CACHE BOOL "" FORCE)
set(BLAZE_INSTALL OFF CACHE BOOL "" FORCE)
set(BLAZE_TESTS OFF CACHE BOOL "" FORCE)
set(BLAZE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(BLAZE_CONTRIB OFF CACHE BOOL "" FORCE)
set(BLAZE_DOCS OFF CACHE BOOL "" FORCE)

set(SOURCEMETA_CORE_JSON ON CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_JSONPOINTER ON CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_JSONSCHEMA ON CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_URI ON CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_REGEX ON CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_JSONL OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_YAML OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_EXTENSION_ALTERSCHEMA OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_TESTS OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_DOCS OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_INSTALL OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_CONTRIB_GOOGLETEST OFF CACHE BOOL "" FORCE)
set(SOURCEMETA_CORE_CONTRIB_GOOGLEBENCHMARK OFF CACHE BOOL "" FORCE)

add_subdirectory("${MODSECURITY_SOURCE_DIR}/others/blaze"
  "${CMAKE_CURRENT_BINARY_DIR}/blaze")

add_library(modsecurity_blaze_bridge STATIC bridge.cc)
set_target_properties(modsecurity_blaze_bridge PROPERTIES
  OUTPUT_NAME modsecurity_blaze_bridge
  CXX_STANDARD 20
  CXX_STANDARD_REQUIRED YES
  CXX_EXTENSIONS NO)
target_include_directories(modsecurity_blaze_bridge PRIVATE
  "${MODSECURITY_SOURCE_DIR}")
target_link_libraries(modsecurity_blaze_bridge PRIVATE
  sourcemeta_blaze_compiler
  sourcemeta_blaze_evaluator)
